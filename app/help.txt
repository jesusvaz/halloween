import { createContext, useContext } from "react";
///
///Declarndo varisble Theme que contiene un objeto llamdo colors
/// que tiene 4 vriables de typo string
/// cada propiedad es de tipo stringn en este caso
type Theme = {
    colors: {
        primary: string
        secondary: string
        terciary: string
        lang:string
    }
}

/// para inicializar la varible defaultTheme
/// se especifica el tipo de variable Theme
/// y entonces se inicializan las variables con los valores
/// correspondientes
const defaultTheme: Theme = {
    colors: {
        primary: "#007bff",
        secondary: "#6c757d",
        terciary: 'yellow',
        lang: 'en'
    }
}

/// crear contexto pasandole el tipo de vcariable
const ThemeContext = createContext(defaultTheme);

/// crear provider
export const ThemeProvider =({ children } : { children: React.ReactNode}) => {
    return (
        // crear provider pasandole el estado qu s vs utilizar en todas las pahuinas
        // que se representan por la variab le children
        // las s cuales tendran acceso  alas propiedades dentro de defaulTheme
        // que en este caso son:
        // primary: "#007bff",
        // secondary: "#6c757d",
        // terciary: 'yellow',
        // lang: 'en'
        <ThemeContext.Provider value={defaultTheme}>
            {children}
        </ThemeContext.Provider>
    )
}




export const useTheme = () => useContext(ThemeContext);

INDICE
1. Parts of Speech
Strategy: Use personal interests or hobbies. Example Activity: Ask students to create sentences about something they love. 
For example:

Nouns: "My favorite musician is Taylor Swift."
Verbs: "I play soccer every weekend."
Adjectives: "That game was exciting."
Ask: "What are your favorite hobbies or activities? Let’s create sentences using nouns, verbs, and adjectives related to them."

2. Sentence Structure
Strategy: Relate to current events or personal goals. Example Activity: Have students write about their weekend plans.

Simple sentence: "I will visit my grandma."
Compound sentence: "I will visit my grandma, and then I’ll watch a movie."
Complex sentence: "Although it might rain, I will visit my grandma."
Ask: "What are your plans for the weekend? Can you describe them using different sentence structures?"

3. Tenses
Strategy: Link to storytelling or journaling. Example Activity: Ask students to tell a story using different tenses.

Past Simple: "Last summer, I went to the beach."
Present Continuous: "I am reading a great book right now."
Future Simple: "Next year, I will travel to Europe."
Ask: "Can you tell me a story about something you did, something you’re doing now, and something you will do in the future?"

4. Articles
Strategy: Connect to favorite things. Example Activity: Discuss favorite movies, books, or places.

Definite article: "The book I’m reading is amazing."
Indefinite article: "I saw a great movie last night."
Ask: "What’s a movie or book you love? How would you describe it using articles?"

5. Modal Verbs
Strategy: Use scenarios students care about. Example Activity: Create rules or advice for a club or a game.

Must: "You must be on time."
Can: "You can choose your own team."
Ask: "If you were in charge of a club, what rules or advice would you give? Use modal verbs to express them."

6. Conditional Sentences
Strategy: Use dreams or hypothetical situations. Example Activity: Ask students about their dreams or "what if" scenarios.

First Conditional: "If I study hard, I will pass the exam."
Second Conditional: "If I won the lottery, I would buy a house."
Third Conditional: "If I had known about the concert, I would have gone."
Ask: "What would you do if you could do anything? Let’s create sentences about it."

7. Passive Voice
Strategy: Connect to news or history. Example Activity: Use headlines or historical events.

Active: "The chef cooked the meal."
Passive: "The meal was cooked by the chef."
Ask: "Can you rewrite this news headline or historical fact in passive voice?"

8. Questions and Negations
Strategy: Use pop culture or personal experiences. Example Activity: Discuss a favorite movie or TV show.

Yes/No: "Did you watch the latest episode?"
WH-: "Where do you think the character will go next?"
Negative: "I don’t think that character will survive."
Ask: "What’s your favorite show? Let’s ask some questions and make predictions about it."

9. Prepositions
Strategy: Relate to familiar spaces. Example Activity: Describe the layout of their room or a favorite place.

Place: "The desk is next to the bed."
Time: "The party starts at 7 PM."
Ask: "Can you describe your room using prepositions? What’s next to your bed?"

10. Comparatives and Superlatives
Strategy: Compare personal choices. Example Activity: Ask students to compare favorite sports, foods, or celebrities.

Comparative: "Soccer is more popular than tennis."
Superlative: "Pizza is the most delicious food."
Ask: "What’s better: soccer or basketball? Let’s compare them using comparatives and superlatives."

11. Pronouns
Strategy: Use storytelling or personalization. Example Activity: Have students tell a story about themselves or someone they know.

Subject Pronoun: "She loves to dance."
Object Pronoun: "I gave him a gift."
Ask: "Can you tell me a story about someone you admire using pronouns?"

12. Reported Speech
Strategy: Connect to conversations they’ve had. Example Activity: Ask students to report what a friend or family member said.

Direct: "He said, 'I am tired.'"
Indirect: "He said that he was tired."
Ask: "What did your friend tell you yesterday? Can you report it in indirect speech?"

13. Phrasal Verbs
Strategy: Use in context of daily activities. Example Activity: Ask students to describe their daily routine using phrasal verbs.

"I woke up early."
"She looked after her little brother."
Ask: "What’s your morning routine? Can you use some phrasal verbs to describe it?"

14. Gerunds and Infinitives
Strategy: Link to hobbies or aspirations. Example Activity: Discuss what they enjoy doing or what they want to do in the future.

Gerund: "I enjoy playing video games."
Infinitive: "I want to learn how to play guitar."
Ask: "What do you enjoy doing in your free time? What do you want to learn in the future?"

15. Conjunctions and Linking Words
Strategy: Connect to storytelling or debates. Example Activity: Create a story or have a debate using conjunctions.

Coordinating: "I wanted to go, but it started raining."
Subordinating: "I stayed home because it was raining."
Ask: "Let’s create a story or debate. How would you link your ideas using conjunctions?"

By personalizing the topics and making them relevant to the students' interests and experiences, you'll help them create vivid examples in their minds, making grammar both engaging and memorable.
